**Full implementation:**

```typescript
import { Extension } from '@tiptap/core'
import { Plugin, PluginKey } from '@tiptap/pm/state'
import { DecorationSet } from '@tiptap/pm/view'
import type { PendingEdit, WorkspaceAction } from '../../types/workspace'
import { buildDecorations } from './decorations'

export const PendingEditsPluginKey = new PluginKey<DecorationSet>('pendingEdits')

export interface PendingEditsOptions {
  sectionKey: string
  getState: () => PendingEdit[]
  dispatch: (action: WorkspaceAction) => void
}

export const PendingEditsPlugin = Extension.create<PendingEditsOptions>({
  name: 'pendingEdits',

  addOptions() {
    return {
      sectionKey: '',
      getState: () => [],
      dispatch: () => {},
    }
  },

  addProseMirrorPlugins() {
    const { sectionKey, getState, dispatch } = this.options

    return [
      new Plugin<DecorationSet>({
        key: PendingEditsPluginKey,

        state: {
          init(_, editorState) {
            const pending = getState()
            if (pending.length === 0) return DecorationSet.empty
            return buildDecorations(pending, editorState.doc, dispatch, sectionKey)
          },

          apply(tr, oldDecoSet, _oldState, newState) {
            // CRITICAL: Always map first — keeps decoration positions alive through user edits
            const mapped = oldDecoSet.map(tr.mapping, newState.doc)

            // Skip rebuild during Accept transaction (content already committed, decoration removed by refresh)
            if (tr.getMeta(PendingEditsPluginKey) === 'skip-refresh') {
              return mapped
            }

            // Staleness detection — only when doc changed AND pending edits exist
            if (tr.docChanged) {
              const pending = getState().filter((c) => c.resolution === 'pending')
              if (pending.length > 0) {
                const staleIds: string[] = []
                for (const change of pending) {
                  if (!change.paragraph_id) continue
                  let found = false
                  newState.doc.descendants((node) => {
                    if (node.attrs?.id === change.paragraph_id) {
                      found = true
                      return false // stop traversal
                    }
                  })
                  if (!found) staleIds.push(change.paragraph_id)
                }

                if (staleIds.length > 0) {
                  // Dispatch React action — deferred from ProseMirror transaction cycle
                  // Using queueMicrotask to avoid dispatching React state during ProseMirror update
                  queueMicrotask(() => {
                    dispatch({ type: 'AUTO_REJECT_STALE_EDITS', payload: { section_key: sectionKey, stale_ids: staleIds } })
                  })

                  // Rebuild without stale items immediately (optimistic visual)
                  const withoutStale = pending.filter((c) => !staleIds.includes(c.paragraph_id))
                  if (withoutStale.length === 0) return DecorationSet.empty
                  return buildDecorations(withoutStale, newState.doc, dispatch, sectionKey)
                }
              }
            }

            // Workspace state changed — rebuild from scratch (triggered by SET_PENDING_EDITS, ACCEPT, REJECT)
            if (tr.getMeta(PendingEditsPluginKey) === 'refresh') {
              const pending = getState().filter((c) => c.resolution === 'pending')
              if (pending.length === 0) return DecorationSet.empty
              return buildDecorations(pending, newState.doc, dispatch, sectionKey)
            }

            return mapped
          },
        },

        props: {
          decorations(state) {
            return PendingEditsPluginKey.getState(state)
          },
        },
      }),
    ]
  },
})
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| ProseMirror plugin → React context | Plugin dispatches WorkspaceAction; React reducer commits; bidirectional but one-way at a time |
| Ghost widget → document content | Widget decorations are view-only; document content must never include ghost text |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14.2-03-01 | Tampering | Ghost content in editor.getHTML() | mitigate | `ghostContentLeakDetected()` guard in decorations.tsx; caller (Plan 05) must check before dispatching SET_PENDING_EDITS. If leak detected: block dispatch, log error with proposal_id + section_key (AI-SPEC online guardrail 1) |
| T-14.2-03-02 | Tampering | Stale ProseMirror dispatch inside plugin apply | mitigate | `queueMicrotask` defers React dispatch outside ProseMirror's transaction commit cycle; prevents synchronous cross-dispatch loops |
| T-14.2-03-03 | Denial of Service | Decoration staleness walk on every transaction | accept | Walk gated on `tr.docChanged && pending.length > 0`; for typical proposals (5–20 paragraphs), per-transaction cost is ~0.1ms — negligible. Doc size bounded by proposal structure. |
| T-14.2-03-04 | Tampering | Accept transaction ordering | mitigate | `setMeta('skip-refresh')` on content replacement transaction prevents decoration rebuild during content mutation; `setMeta('refresh')` on subsequent transaction rebuilds cleanly. Two-phase pattern prevents partial decoration state. |

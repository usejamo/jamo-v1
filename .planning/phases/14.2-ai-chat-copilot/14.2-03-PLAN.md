```typescript
function computeAnchorHash(paragraphEl: Element, paragraphId: string): string {
  const normalizedText = (paragraphEl.textContent ?? '')
    .trim()
    .replace(/\s+/g, ' ')
  const rawHtml = paragraphEl.innerHTML
  const combined = normalizedText + '|' + paragraphId + '|' + rawHtml
  // Simple deterministic hash (full SHA-256 via Web Crypto is async — use djb2 for sync)
  let h = 5381
  for (let i = 0; i < combined.length; i++) {
    h = ((h << 5) + h) ^ combined.charCodeAt(i)
  }
  return (h >>> 0).toString(16).slice(0, 8)
}
```

**Plugin state protocol — no dispatch from apply():**
The plugin must NOT call `dispatch(workspaceAction)` inside `apply()`. Instead, the plugin's state carries metadata that Plan 05's `SectionEditorBlock` reads via `PendingEditsPluginKey.getState(editorState)`. The SectionEditorBlock effect layer then dispatches workspace actions.

However, to maintain backward compatibility with the Plan 01 stub (which passes `dispatch` as an option), the plugin MAY use `queueMicrotask(() => dispatch(...))` when stale anchors are detected — this defers the dispatch outside the ProseMirror transaction cycle. Plan 05 will add a proper effect-layer subscription.

```typescript
import { Extension } from '@tiptap/core'
import { Plugin, PluginKey } from '@tiptap/pm/state'
import { DecorationSet } from '@tiptap/pm/view'
import type { PendingEdit, WorkspaceAction } from '../../types/workspace'
import { buildDecorations } from './decorations'

export const PendingEditsPluginKey = new PluginKey<DecorationSet>('pendingEdits')

export interface PendingEditsOptions {
  sectionKey: string
  getState: () => PendingEdit[]
  dispatch: (action: WorkspaceAction) => void
}

export const PendingEditsPlugin = Extension.create<PendingEditsOptions>({
  name: 'pendingEdits',

  addOptions() {
    return { sectionKey: '', getState: () => [], dispatch: () => {} }
  },

  addProseMirrorPlugins() {
    const { sectionKey, getState, dispatch } = this.options

    return [
      new Plugin<DecorationSet>({
        key: PendingEditsPluginKey,

        state: {
          init(_, editorState) {
            const pending = getState()
            if (pending.length === 0) return DecorationSet.empty
            return buildDecorations(pending, editorState.doc, dispatch, sectionKey)
          },

          apply(tr, oldDecoSet, _oldState, newState) {
            // Always map first — keeps decoration positions alive through user edits
            const mapped = oldDecoSet.map(tr.mapping, newState.doc)

            // Skip rebuild during Accept/Batch transaction (content being committed)
            if (tr.getMeta(PendingEditsPluginKey) === 'skip-refresh') {
              return mapped
            }

            // Staleness detection — only when doc changed AND pending edits exist
            // CRITICAL: dispatch is deferred via queueMicrotask — NEVER called inline inside apply()
            // Plan 05 SectionEditorBlock reads PendingEditsPluginKey.getState() to observe stale state
            if (tr.docChanged) {
              const pending = getState().filter((c) => c.resolution === 'pending')
              if (pending.length > 0) {
                const staleIds: string[] = []
                for (const change of pending) {
                  if (!change.paragraph_id) continue
                  let found = false
                  newState.doc.descendants((node) => {
                    if (node.attrs?.id === change.paragraph_id) { found = true; return false }
                  })
                  if (!found) staleIds.push(change.paragraph_id)
                }

                if (staleIds.length > 0) {
                  // Defer workspace dispatch OUTSIDE the ProseMirror transaction commit cycle.
                  // SectionEditorBlock effect layer (Plan 05) also watches plugin state for stale_ids.
                  queueMicrotask(() => {
                    dispatch({
                      type: 'AUTO_REJECT_STALE_EDITS',
                      payload: { section_key: sectionKey, stale_ids: staleIds },
                    })
                  })
                  const withoutStale = pending.filter((c) => !staleIds.includes(c.paragraph_id))
                  if (withoutStale.length === 0) return DecorationSet.empty
                  return buildDecorations(withoutStale, newState.doc, dispatch, sectionKey)
                }
              }
            }

            // Workspace state changed (SET_PENDING_EDITS, ACCEPT, REJECT etc.) — rebuild from scratch
            if (tr.getMeta(PendingEditsPluginKey) === 'refresh') {
              const pending = getState().filter((c) => c.resolution === 'pending')
              if (pending.length === 0) return DecorationSet.empty
              return buildDecorations(pending, newState.doc, dispatch, sectionKey)
            }

            return mapped
          },
        },

        props: {
          decorations(state) {
            return PendingEditsPluginKey.getState(state)
          },
        },
      }),
    ]
  },
})
```

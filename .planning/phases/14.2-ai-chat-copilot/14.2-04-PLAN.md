```typescript
case 'SET_PENDING_EDITS': {
  const { section_key, edits } = action.payload
  const section = state.sections[section_key]
  if (!section) return state
  return {
    ...state,
    sections: {
      ...state.sections,
      [section_key]: {
        ...section,
        pending_edits: edits.map((e) => ({ ...e, resolution: 'pending' as const })),
      },
    },
  }
}

case 'ACCEPT_PENDING_EDIT': {
  const { section_key, paragraph_id } = action.payload
  const section = state.sections[section_key]
  if (!section) return state
  const updatedEdits = section.pending_edits.map((e) =>
    e.paragraph_id === paragraph_id ? { ...e, resolution: 'accepted' as const } : e
  )
  // Side effect: persist resolutions to DB (D-14/D-15 — fire-and-forget)
  const changed = updatedEdits.find((e) => e.paragraph_id === paragraph_id)
  if (changed) {
    void persistResolutions(changed.message_id, buildResolutionMap(updatedEdits))
  }
  return {
    ...state,
    sections: {
      ...state.sections,
      [section_key]: { ...section, pending_edits: updatedEdits },
    },
  }
}

case 'REJECT_PENDING_EDIT': {
  const { section_key, paragraph_id } = action.payload
  const section = state.sections[section_key]
  if (!section) return state
  const updatedEdits = section.pending_edits.map((e) =>
    e.paragraph_id === paragraph_id ? { ...e, resolution: 'rejected' as const } : e
  )
  const changed = updatedEdits.find((e) => e.paragraph_id === paragraph_id)
  if (changed) {
    void persistResolutions(changed.message_id, buildResolutionMap(updatedEdits))
  }
  return {
    ...state,
    sections: {
      ...state.sections,
      [section_key]: { ...section, pending_edits: updatedEdits },
    },
  }
}

case 'CLEAR_PENDING_EDITS': {
  const { section_key } = action.payload
  const section = state.sections[section_key]
  if (!section) return state
  return {
    ...state,
    sections: {
      ...state.sections,
      [section_key]: { ...section, pending_edits: [] },
    },
  }
}

case 'AUTO_REJECT_STALE_EDITS': {
  const { section_key, stale_ids } = action.payload
  const section = state.sections[section_key]
  if (!section) return state
  const updatedEdits = section.pending_edits.map((e) =>
    stale_ids.includes(e.paragraph_id)
      ? { ...e, resolution: 'auto_rejected_stale' as const }
      : e
  )
  // Persist auto-rejections as a batch (single call per message_id)
  const staleMessages = new Set(
    updatedEdits.filter((e) => stale_ids.includes(e.paragraph_id)).map((e) => e.message_id)
  )
  for (const messageId of staleMessages) {
    const relevantEdits = updatedEdits.filter((e) => e.message_id === messageId)
    void persistResolutions(messageId, buildResolutionMap(relevantEdits))
  }
  return {
    ...state,
    sections: {
      ...state.sections,
      [section_key]: { ...section, pending_edits: updatedEdits },
    },
  }
}

case 'BATCH_ACCEPT_PENDING_EDITS': {
  // D-03: Batch Accept — reducer marks all pending edits as 'accepted' in one state update.
  // SectionEditorBlock (Plan 05) watches the resulting state change and builds ONE chained
  // ProseMirror transaction for all newly-accepted edits — one undo step, not N.
  const { section_key, edits } = action.payload
  const section = state.sections[section_key]
  if (!section) return state

  // Only process edits that are currently pending — skip already-resolved edits
  const pendingIds = new Set(
    edits.filter((e) => e.resolution === 'pending').map((e) => e.paragraph_id)
  )

  const updatedEdits = section.pending_edits.map((e) =>
    pendingIds.has(e.paragraph_id) ? { ...e, resolution: 'accepted' as const } : e
  )

  // Single persist call for the full batch (one DB write per batch, not one per edit)
  if (updatedEdits.length > 0) {
    // Group by message_id (all edits from same propose_edit call share a message_id)
    const messageGroups = new Map<string, typeof updatedEdits>()
    for (const e of updatedEdits) {
      if (!messageGroups.has(e.message_id)) messageGroups.set(e.message_id, [])
      messageGroups.get(e.message_id)!.push(e)
    }
    for (const [messageId, groupEdits] of messageGroups) {
      void persistResolutions(messageId, buildResolutionMap(groupEdits))
    }
  }

  return {
    ...state,
    sections: {
      ...state.sections,
      [section_key]: { ...section, pending_edits: updatedEdits },
    },
  }
}
```

```typescript
import { supabase } from '../lib/supabase'

async function persistResolutions(
  messageId: string,
  resolutions: Record<string, string>,
): Promise<void> {
  try {
    const { data } = await supabase
      .from('proposal_chats')
      .select('tool_data')
      .eq('id', messageId)
      .single()
    if (!data?.tool_data) return
    const updatedToolData = {
      ...data.tool_data,
      state: { ...(data.tool_data.state ?? {}), resolutions },
    }
    await supabase
      .from('proposal_chats')
      .update({ tool_data: updatedToolData })
      .eq('id', messageId)
  } catch {
    // D-15: fire-and-forget — silent fail, no toast here (toast lives in AIChatPanel)
  }
}
```

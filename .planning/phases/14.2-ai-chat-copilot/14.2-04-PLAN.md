case 'SET_PENDING_EDITS': {

  if (!section) return state
  const updatedEdits = section.pending_edits.map((e) =>

    void persistResolutions(changed.message_id, resolutions)

    },
  }

case 'REJECT_PENDING_EDIT': {
  const { section_key, paragraph_id } = action.payload
  const section = state.sections[section_key]
  if (!section) return state
  const updatedEdits = section.pending_edits.map((e) =>
    e.paragraph_id === paragraph_id ? { ...e, resolution: 'rejected' as const } : e

  const changed = updatedEdits.find((e) => e.paragraph_id === paragraph_id)
  if (changed) {
    const resolutions = buildResolutionMap(updatedEdits)
    void persistResolutions(changed.message_id, resolutions)
  }
  return {
    ...state,
    sections: {
      ...state.sections,
      [section_key]: { ...section, pending_edits: updatedEdits },
    },
  }
}

case 'CLEAR_PENDING_EDITS': {
  const { section_key } = action.payload
  const section = state.sections[section_key]
  if (!section) return state
  return {
    ...state,
    sections: {
      ...state.sections,
      [section_key]: { ...section, pending_edits: [] },
    },
  }

case 'AUTO_REJECT_STALE_EDITS': {
  const { section_key, stale_ids } = action.payload
  const section = state.sections[section_key]
  if (!section) return state
  const updatedEdits = section.pending_edits.map((e) =>
    stale_ids.includes(e.paragraph_id)
      ? { ...e, resolution: 'auto_rejected_stale' as const }
      : e
  )
  // Persist auto-rejections as a batch (single call with all stale resolutions)
  const staleMessages = new Set(
    updatedEdits.filter((e) => stale_ids.includes(e.paragraph_id)).map((e) => e.message_id)
  )
  for (const messageId of staleMessages) {
    const relevantEdits = updatedEdits.filter((e) => e.message_id === messageId)
    void persistResolutions(messageId, buildResolutionMap(relevantEdits))
  }
  return {
    ...state,
    sections: {
      ...state.sections,
      [section_key]: { ...section, pending_edits: updatedEdits },
    },
  }

**Preferred option (simpler):** Add a direct Supabase write inside `SectionWorkspaceContext.tsx` — import the Supabase client and write directly:

```typescript
import { supabase } from '../lib/supabase'

async function persistResolutions(
  messageId: string,
  resolutions: Record<string, string>,
): Promise<void> {
  try {
    // Read current tool_data state, merge resolutions, write back
    const { data } = await supabase
      .from('proposal_chats')
      .select('tool_data')
      .eq('id', messageId)
      .single()
    if (!data?.tool_data) return
    const updatedToolData = {
      ...data.tool_data,
      state: { ...(data.tool_data.state ?? {}), resolutions },
    }
    await supabase
      .from('proposal_chats')
      .update({ tool_data: updatedToolData })
      .eq('id', messageId)
  } catch {
    // D-15: fire-and-forget — silent fail, no toast here (toast lives in AIChatPanel)
  }
}
```

This is a fire-and-forget pattern — `void persistResolutions(...)` in the reducer means the reducer returns synchronously; the async write happens in the background. D-15 requires silent failure here; UI error toasts are surfaced by the `AIChatPanel`'s existing error handling on Supabase writes.

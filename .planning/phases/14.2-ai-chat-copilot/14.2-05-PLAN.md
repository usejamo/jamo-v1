**2. Add a useEffect that fires the plugin refresh on pending_edits changes:**

After the `useEditor` call, add:

```typescript
const pendingEdits = workspaceState.sections[sectionKey]?.pending_edits ?? []
const pendingEditsLengthRef = useRef<number>(pendingEdits.length)
const pendingEditsIdsRef = useRef<string>('')

useEffect(() => {
  if (!editor) return
  const currentIds = pendingEdits.map((e) => e.paragraph_id + e.resolution).join(',')
  // Only fire refresh if the set or resolutions changed (prevent loop — Pitfall 2)
  if (
    currentIds === pendingEditsIdsRef.current &&
    pendingEdits.length === pendingEditsLengthRef.current
  ) return

  pendingEditsLengthRef.current = pendingEdits.length
  pendingEditsIdsRef.current = currentIds

  editor.view.dispatch(
    editor.state.tr.setMeta(PendingEditsPluginKey, 'refresh')
  )
}, [editor, pendingEdits])
```

The `currentIds` string encodes both which changes exist AND their current resolution — so any Accept/Reject also triggers a refresh. This removes accepted/rejected ghosts cleanly.

**3. Add a useEffect that performs the ProseMirror content replacement when ACCEPT_PENDING_EDIT fires:**

When a ghost is accepted, the workspace reducer updates `resolution` to `'accepted'`. The plugin refresh removes the ghost decoration. But the actual paragraph content replacement must happen here via a ProseMirror transaction:

```typescript
const prevResolutionsRef = useRef<Record<string, string>>({})

useEffect(() => {
  if (!editor) return
  const currentEdits = workspaceState.sections[sectionKey]?.pending_edits ?? []
  
  for (const edit of currentEdits) {
    const prevRes = prevResolutionsRef.current[edit.paragraph_id]
    if (prevRes !== 'accepted' && edit.resolution === 'accepted' && edit.operation !== 'delete') {
      // Perform the ProseMirror content replacement
      const newHtml = edit.after_html ?? ''
      if (!newHtml) continue
      
      // Find anchor position
      let anchorFrom = -1
      let anchorTo = -1
      editor.state.doc.descendants((node, pos) => {
        if (node.attrs?.id === edit.paragraph_id) {
          anchorFrom = pos; anchorTo = pos + node.nodeSize; return false
        }
      })
      
      if (anchorFrom === -1) continue // anchor no longer exists — stale, skip
      
      // Parse new content and replace
      const newNode = editor.state.schema.nodeFromJSON(
        // Use editor's parseHTML helper to create a proper ProseMirror node from the HTML fragment
        editor.commands.setContent // not quite — use DOMParser approach below
      )
      
      // Use TipTap's insertContentAt with aiApplied meta:
      editor.chain()
        .command(({ tr }) => {
          tr.setMeta('aiApplied', true)
          tr.setMeta('addToHistory', true)
          tr.setMeta(PendingEditsPluginKey, 'skip-refresh')
          return true
        })
        .deleteRange({ from: anchorFrom, to: anchorTo })
        .insertContentAt(anchorFrom, newHtml)
        .run()
    }
    
    if (prevRes !== 'accepted' && edit.resolution === 'accepted' && edit.operation === 'delete') {
      // For deletes, Accept removes the paragraph
      let anchorFrom = -1; let anchorTo = -1
      editor.state.doc.descendants((node, pos) => {
        if (node.attrs?.id === edit.paragraph_id) { anchorFrom = pos; anchorTo = pos + node.nodeSize; return false }
      })
      if (anchorFrom === -1) continue
      editor.chain()
        .command(({ tr }) => { tr.setMeta('aiApplied', true); tr.setMeta('addToHistory', true); tr.setMeta(PendingEditsPluginKey, 'skip-refresh'); return true })
        .deleteRange({ from: anchorFrom, to: anchorTo })
        .run()
    }
  }
  
  // Update prevResolutions for next comparison
  prevResolutionsRef.current = Object.fromEntries(currentEdits.map((e) => [e.paragraph_id, e.resolution]))
}, [editor, workspaceState.sections[sectionKey]?.pending_edits])
```

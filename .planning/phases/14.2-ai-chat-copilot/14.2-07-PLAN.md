   ```typescript
   const [pendingActions, setPendingActions] = useState<PendingActionItem[]>([])
   const [activeTask, setActiveTask] = useState<ActiveTask | null>(null)
   
   // Load initial state from chat_sessions on mount
   useEffect(() => {
     if (!proposalId) return
     supabase
       .from('chat_sessions')
       .select('pending_actions, active_task')
       .eq('proposal_id', proposalId)
       .single()
       .then(({ data }) => {
         if (data?.pending_actions) setPendingActions(data.pending_actions as PendingActionItem[])
         if (data?.active_task) setActiveTask(data.active_task as ActiveTask)
       })
   }, [proposalId])
   
   // Realtime subscription for live pending_actions + active_task updates
   useEffect(() => {
     if (!proposalId) return
     const channel = supabase
       .channel(`chat_sessions:${proposalId}`)
       .on('postgres_changes', {
         event: 'UPDATE', schema: 'public', table: 'chat_sessions',
         filter: `proposal_id=eq.${proposalId}`,
       }, (payload) => {
         const row = payload.new as { pending_actions?: PendingActionItem[]; active_task?: ActiveTask | null }
         if (row.pending_actions !== undefined) setPendingActions(row.pending_actions ?? [])
         if ('active_task' in row) setActiveTask(row.active_task ?? null)
       })
       .subscribe()
     return () => { supabase.removeChannel(channel) }
   }, [proposalId])
   ```

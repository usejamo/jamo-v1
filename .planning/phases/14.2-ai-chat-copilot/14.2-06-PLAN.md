```typescript
import Anthropic from "npm:@anthropic-ai/sdk"
import { createClient } from "npm:@supabase/supabase-js@2"
import { z } from "npm:zod@^3"
import { crypto } from "jsr:@std/crypto"

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

// ── Constants (D-28: all cap values in one place, tunable) ──────────────────
const QUEUE_CAP = 8
const TIER_CAPS = { compliance: 4, conflict: 2, gap: 2, missing: 2 } as const
const RATE_LIMIT_MS = 3000   // D-32: 30s cooldown was revised to 3s in D-30
const PROPOSAL_COOLDOWN_MS = 30_000  // D-32: per-proposal 30s cooldown

// ── PendingAction Zod schema (matches PendingActionItem interface from chat.ts) ──
const PendingActionSchema = z.object({
  id: z.string().uuid(),
  type: z.enum(['gap', 'conflict', 'compliance', 'missing']),
  section_key: z.string().min(1),
  title: z.string().min(1).max(200),
  description: z.string().min(1).max(500),
  priority: z.union([z.literal(1), z.literal(2), z.literal(3), z.literal(4)]),
  cta_label: z.string().min(1),
  cta_tool: z.enum(['propose_edit', 'check_regulatory_compliance', 'answer_with_citations']),
  cta_payload: z.record(z.unknown()),
})

type PendingAction = z.infer<typeof PendingActionSchema>

// ── Analysis system prompt ──────────────────────────────────────────────────
const ANALYSIS_SYSTEM_PROMPT = `You are a CRO proposal quality analyst. You receive a JSON array of proposal section summaries and return a JSON array of quality issues.

Return ONLY a valid JSON array. No explanation, no markdown. Each item must match:
{
  "id": "<uuid>",
  "type": "gap" | "conflict" | "compliance" | "missing",
  "section_key": "<key from input>",
  "title": "<Section Name> — <short issue description>",
  "description": "<one sentence explaining the issue>",
  "priority": 1 | 2 | 3 | 4,
  "cta_label": "Fix it" | "Draft it" | "Check it",
  "cta_tool": "propose_edit" | "check_regulatory_compliance" | "answer_with_citations",
  "cta_payload": {}
}

Priority rules (D-18 — fixed, not Haiku-scored):
- compliance issues: priority 1
- conflict issues: priority 2  
- gap issues: priority 3
- missing issues: priority 4

Type definitions (D-20 — distinction is mandatory):
- "gap": section exists but content is incomplete or thin
- "missing": a required section that has not been started (empty/absent)
- "conflict": cross-section inconsistency (numbers, dates, scope)
- "compliance": regulatory language concern

Title format (D-29): "[Section Name] — [description]". Section name must be visible, not in tooltip.

Few-shot examples:
INPUT: [{"key":"eligibility","title":"Eligibility Criteria","excerpt":"TBD"}]
OUTPUT: [{"id":"...","type":"missing","section_key":"eligibility","title":"Eligibility Criteria — section not started","description":"Eligibility Criteria section has no content and must be drafted before submission.","priority":4,"cta_label":"Draft it","cta_tool":"propose_edit","cta_payload":{"section_key":"eligibility"}}]

INPUT: [{"key":"budget","title":"Budget","excerpt":"18 months total"},{"key":"timeline","title":"Timeline","excerpt":"24-month study duration"}]
OUTPUT: [{"id":"...","type":"conflict","section_key":"budget","title":"Budget — duration conflicts with Timeline","description":"Budget states 18 months but Timeline section specifies 24 months. One must be corrected.","priority":2,"cta_label":"Fix it","cta_tool":"propose_edit","cta_payload":{"section_key":"budget"}}]`

// ── Main handler ─────────────────────────────────────────────────────────────
Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders })

  const supabaseUrl = Deno.env.get('SUPABASE_URL')!
  const supabaseKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
  const supabase = createClient(supabaseUrl, supabaseKey)

  let proposal_id: string, org_id: string, sections: Array<{key: string; title: string; content: string}>
  let run_id: string  // D-34: race condition guard

  try {
    const body = await req.json()
    proposal_id = body.proposal_id
    org_id = body.org_id
    sections = body.sections
    run_id = body.run_id ?? crypto.randomUUID()
    if (!proposal_id || !org_id || !Array.isArray(sections)) {
      return new Response(JSON.stringify({ error: 'missing required fields' }), { status: 400, headers: corsHeaders })
    }
  } catch {
    return new Response(JSON.stringify({ error: 'invalid JSON' }), { status: 400, headers: corsHeaders })
  }

  // D-32: Per-proposal 30s cooldown
  const { data: session } = await supabase
    .from('chat_sessions')
    .select('last_updated')
    .eq('proposal_id', proposal_id)
    .single()

  if (session?.last_updated) {
    const age = Date.now() - new Date(session.last_updated).getTime()
    if (age < PROPOSAL_COOLDOWN_MS) {
      return new Response(null, { status: 429, headers: corsHeaders })
    }
  }

  // Build truncated section summaries (AI-SPEC: 300 chars max per section)
  const summaries = sections.map((s) => ({
    key: s.key,
    title: s.title,
    // Strip HTML tags, truncate to 300 chars (AI-SPEC: send summaries only)
    excerpt: s.content.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ').trim().slice(0, 300),
  }))

  const anthropic = new Anthropic({ apiKey: Deno.env.get('ANTHROPIC_API_KEY')! })

  let pendingActions: PendingAction[] = []
  try {
    const response = await anthropic.messages.create({
      model: 'claude-haiku-4-5-20251001',  // AI-SPEC: Haiku ONLY for gap analysis — NEVER Sonnet
      max_tokens: 1024,
      temperature: 0,  // deterministic classification output
      system: ANALYSIS_SYSTEM_PROMPT,
      messages: [{ role: 'user', content: JSON.stringify(summaries) }],
    })

    const textBlock = response.content.find((b) => b.type === 'text')
    const raw = JSON.parse(textBlock?.text ?? '[]')

    // Zod validate before writing to DB — reject invalid items silently (AI-SPEC)
    const validated = z.array(PendingActionSchema).safeParse(raw)
    if (validated.success) {
      // Apply priority-based type ordering and tier caps (D-18, D-26)
      const byType = (t: string) => ({ compliance: 1, conflict: 2, gap: 3, missing: 4 }[t] ?? 5)
      const sorted = validated.data.sort((a, b) => byType(a.type) - byType(b.type))

      const counts = { compliance: 0, conflict: 0, gap: 0, missing: 0 }
      pendingActions = sorted.filter((item) => {
        const tier = item.type as keyof typeof TIER_CAPS
        if (counts[tier] >= TIER_CAPS[tier]) return false
        counts[tier]++
        return true
      }).slice(0, QUEUE_CAP)
    }
    // On parse failure: pendingActions stays [] — empty queue, no user error
  } catch {
    // Haiku call failure or JSON parse error — return empty queue silently
    pendingActions = []
  }

  // D-34: Write only if this run_id is still the latest
  // Upsert with run_id tag — client discards stale responses
  await supabase
    .from('chat_sessions')
    .upsert({
      proposal_id,
      org_id,
      pending_actions: pendingActions,
      last_updated: new Date().toISOString(),
    }, { onConflict: 'proposal_id' })

  return new Response(
    JSON.stringify({ ok: true, run_id, action_count: pendingActions.length }),
    { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
  )
})
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Frontend → analyze-proposal-gaps | Client sends section content — edge function processes and writes to DB; no client-controlled DB writes |
| Haiku output → pending_actions | Haiku returns JSON that is Zod-validated before any DB write |
| active_task → AI context | active_task injected into Haiku/Sonnet context — must not inject stale or cross-user state |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-14.2-06-01 | Tampering | Haiku output injection into pending_actions | mitigate | Zod `PendingActionSchema` validates every item before DB write; invalid items are dropped, not written. Schema enforces enum types and string length limits. |
| T-14.2-06-02 | Elevation of Privilege | cross-proposal active_task read | mitigate | `chat_sessions` fetched by `proposal_id` (from request body, validated). RLS on `chat_sessions` enforces `org_id` match — user cannot read another org's `active_task`. |
| T-14.2-06-03 | Denial of Service | Haiku flooding without rate limit | mitigate | 30s per-proposal cooldown enforced in edge function via `last_updated` check. Returns HTTP 429 on violation. Client does not retry on 429 (D-30/D-32). |
| T-14.2-06-04 | Tampering | Stale active_task injected into AI context | mitigate | Auto-expire check (>7 days) clears `active_task` before injection. Client-side resume validation in Plan 07 provides second check. |
| T-14.2-06-05 | Information Disclosure | section content in Haiku request | accept | Section content is sent to Anthropic API (same as existing chat-with-jamo). User already consents to AI processing via product use. Content truncated to 300 chars per section to minimize exposure. |
</threat_model>

```sql
-- Phase 14.2: Enable Realtime on chat_sessions for pending_actions and active_task live updates.
-- Required for ActionQueue (Part B) to update reactively without polling.

-- Idempotent: only add if not already in publication
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_publication_tables
    WHERE pubname = 'supabase_realtime'
      AND tablename = 'chat_sessions'
  ) THEN
    ALTER PUBLICATION supabase_realtime ADD TABLE chat_sessions;
  END IF;
END;
$$;

-- Enable replica identity FULL so Realtime receives full row on UPDATE
-- Required for payload.new to include JSONB columns like pending_actions
ALTER TABLE chat_sessions REPLICA IDENTITY FULL;
```

**`supabase/migrations/20260515000041_chat_sessions_per_user.sql`** (D-44/D-45/D-46/D-47):

Read the existing chat_sessions migration first to understand current PK and columns.

```sql
-- Phase 14.2 — D-44/D-45/D-46/D-47: Per-user chat_sessions schema
-- D-44: Add user_id column to chat_sessions (UUID, references auth.users)
-- D-45: Change PK to composite (proposal_id, user_id) — one session per user per proposal
-- D-46: Create pending_action_dismissals table for persistent dismiss state
-- D-47: Update RLS to include user_id = auth.uid() predicate

-- Step 1: Add user_id column (nullable first for backfill compatibility)
ALTER TABLE chat_sessions
  ADD COLUMN IF NOT EXISTS user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE;

-- Step 2: Backfill existing rows — assign to the proposal owner
-- Joins proposals table to find the creating user. If proposals has no created_by, use a sentinel.
UPDATE chat_sessions cs
SET user_id = p.created_by
FROM proposals p
WHERE cs.proposal_id = p.id
  AND cs.user_id IS NULL
  AND p.created_by IS NOT NULL;

-- Step 3: For any rows still null (proposals without created_by), null is acceptable —
-- they will be claimed on next access. Set NOT NULL constraint after backfill.
-- NOTE: Only add NOT NULL if all rows were backfilled. Check first:
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM chat_sessions WHERE user_id IS NULL
  ) THEN
    ALTER TABLE chat_sessions ALTER COLUMN user_id SET NOT NULL;
  END IF;
END;
$$;

-- Step 4: Drop existing PK (proposal_id) and add composite PK (proposal_id, user_id)
-- Only if the composite PK does not already exist
DO $$
BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM pg_constraint
    WHERE conrelid = 'chat_sessions'::regclass
      AND contype = 'p'
      AND array_length(conkey, 1) = 2
  ) THEN
    -- Drop old single-column PK first
    ALTER TABLE chat_sessions DROP CONSTRAINT IF EXISTS chat_sessions_pkey;
    -- Add composite PK
    ALTER TABLE chat_sessions ADD PRIMARY KEY (proposal_id, user_id);
  END IF;
END;
$$;

-- Step 5: Update RLS policies to include user_id = auth.uid() predicate (D-47)
-- Drop old policies and recreate with user_id filter
DROP POLICY IF EXISTS "chat_sessions_select" ON chat_sessions;
DROP POLICY IF EXISTS "chat_sessions_insert" ON chat_sessions;
DROP POLICY IF EXISTS "chat_sessions_update" ON chat_sessions;
DROP POLICY IF EXISTS "chat_sessions_delete" ON chat_sessions;

-- Select: org_id match AND user owns the session
CREATE POLICY "chat_sessions_select" ON chat_sessions
  FOR SELECT USING (
    org_id = (auth.jwt() ->> 'org_id')::uuid
    AND user_id = auth.uid()
  );

-- Insert: user sets their own user_id
CREATE POLICY "chat_sessions_insert" ON chat_sessions
  FOR INSERT WITH CHECK (
    org_id = (auth.jwt() ->> 'org_id')::uuid
    AND user_id = auth.uid()
  );

-- Update: user can only update their own session
CREATE POLICY "chat_sessions_update" ON chat_sessions
  FOR UPDATE USING (
    org_id = (auth.jwt() ->> 'org_id')::uuid
    AND user_id = auth.uid()
  );

-- Delete: user can only delete their own session
CREATE POLICY "chat_sessions_delete" ON chat_sessions
  FOR DELETE USING (
    org_id = (auth.jwt() ->> 'org_id')::uuid
    AND user_id = auth.uid()
  );

-- Step 6: Create pending_action_dismissals table (D-46)
-- Persists dismiss state so dismissed items survive Realtime re-broadcasts
CREATE TABLE IF NOT EXISTS pending_action_dismissals (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  proposal_id UUID NOT NULL,
  user_id UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  action_id TEXT NOT NULL,  -- matches PendingActionItem.id
  content_hash TEXT,        -- D-22: hash at dismiss time for resurfacing detection
  dismissed_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  UNIQUE (proposal_id, user_id, action_id)
);

ALTER TABLE pending_action_dismissals ENABLE ROW LEVEL SECURITY;

CREATE POLICY "dismissals_select" ON pending_action_dismissals
  FOR SELECT USING (user_id = auth.uid());

CREATE POLICY "dismissals_insert" ON pending_action_dismissals
  FOR INSERT WITH CHECK (user_id = auth.uid());

CREATE POLICY "dismissals_delete" ON pending_action_dismissals
  FOR DELETE USING (user_id = auth.uid());

-- Index for fast lookup by proposal + user
CREATE INDEX IF NOT EXISTS idx_dismissals_proposal_user
```

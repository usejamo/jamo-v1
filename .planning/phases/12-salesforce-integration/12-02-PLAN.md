**useEffect for sf_error (D-15):**
```typescript
useEffect(() => {
  const sfErr = searchParams.get('sf_error')
  if (sfErr) {
    setError(SF_ERROR_COPY[sfErr] ?? SF_ERROR_COPY['unknown'])
    window.history.replaceState(
      {},
      '',
      window.location.pathname +
        window.location.search
          .replace(/[?&]sf_error=[^&]*/g, '')
          .replace(/^&/, '?')
    )
  }
}, []) // run once on mount only
```

**useEffect for DB fetch (D-16 graceful degradation):**
```typescript
useEffect(() => {
  if (!orgId) return
  setFetchLoading(true)
  supabase
    .from('salesforce_connections')
    .select('sf_username, is_sandbox')
    .eq('org_id', orgId)
    .maybeSingle()
    .then(({ data, error }) => {
      // D-16: never throw — error = disconnected state
      setConnection(data ?? null)
      setFetchLoading(false)
    })
}, [orgId])
```

**Connect handler:**
```typescript
const handleConnect = async () => {
  setLoading(true)
  setError(null)
  const { data, error } = await supabase.functions.invoke('salesforce-oauth-initiate', {
    body: { org_id: orgId, is_sandbox: isSandbox }
  })
  if (error || !data?.auth_url) {
    setError(SF_ERROR_COPY['unknown'])
    setLoading(false)
    return
  }
  window.location.href = data.auth_url
  // loading stays true — browser is navigating away
}
```

**Disconnect handler:**
```typescript
const handleDisconnect = async () => {
  setDisconnecting(true)
  const { error } = await supabase.functions.invoke('salesforce-oauth-disconnect', {
    body: { org_id: orgId }
  })
  if (error) {
    setError(SF_ERROR_COPY['unknown'])
    setDisconnecting(false)
    return
  }
  setConnection(null)
  setDisconnecting(false)
}
```
